	\subsection{Особенности раскладки и логистика}
	\begin{itemize}
		\item \textbf{Соотношение б/ж/у:} от <<стандарта>> летних походов Б:Ж:У=20:20:60 ушли в сторону увеличения доли жира. В нашем случае Б:Ж:У=17:25:58.
		\item \textbf{Завтраки:} чередование \textbf{мясных и молочных каш}. В молочных вариантах использовались хлопья.
		\item \textbf{Белок и жир:} \textbf{пеммикан как основной источник белка}; дополнительно \textbf{сало/грудинка}
		(а также колбаса/бекон по дням) как калорийная добавка.
		\item \textbf{Сухари:} часть сухарей, запланированных \textbf{на ужин}, участники сушили на себя сами.
		\item \textbf{Приправы:} широкий набор добавок в чай и горячее (специи, сушёный лимон и пр.);
		\item \textbf{Перевальный шоколад:} отдельный запас «на мораль и экстренную подпитку».
		\item \textbf{Супы и чай:} утром заготавливались \textbf{3 термоса чая и 3 термоса супа по 1~л};
		в обед разливались, \textbf{по термосу на 3 человек}. Это ускорило обед и уменьшило простои на ветру.
	\end{itemize}
	
	\subsection{Отзывы участников}
	\begin{itemize}
		\item \textbf{Пеммикан} получил \textbf{очень высокую оценку} (сытность, вкус, удобство дозировки).
		\item \textbf{Хлопья в молочных кашах} многим \textbf{не понравились} (вкус/консистенция).
		\item \textbf{Сухофрукты утром} практически \textbf{не ели} (лучше переносить на ходовой перекус/вечер).
		\item \textbf{Сладкое:} количество единиц (батончики/печенье) иногда \textbf{не делилось на число участников},
		что усложняло честную раздачу.
	\end{itemize}
	
	\subsection{Замечания и рекомендации начпрода}
	\begin{itemize}
		\item На будущие выходы: заменить хлопья в молочных кашах на крупу/манку/смесь, либо выдавать хлопья как опцию.
		\item Сладкое планировать поштучно кратно численности группы или переводить в развес с дозировкой.
		\item Практику с термосами признать удачной: уменьшает время стоянки и повышает потребление горячего.
	\end{itemize}
